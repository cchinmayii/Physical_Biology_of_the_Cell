\section*{Notation Summary}
% ====================

\begin{center}
\begin{tabular}{ll}
\hline
Symbol & Meaning \\
\hline
$M(t)$ & cell mass at time $t$ \\
$k$ & growth rate or hopping rate, depending on context \\
$m(t)$ & number of mRNA molecules \\
$r$ & mRNA production rate \\
$\gamma$ & mRNA degradation rate \\
$1/\gamma$ & characteristic mRNA lifetime \\
$P(x,t)$ & probability density/probability at position $x$ and time $t$ \\
$a$ & random-walk step length / lattice spacing \\
$\Dt$ & discrete time step \\
$L$ & length of the periodic spatial domain \\
$n$ & Fourier mode number \\
$q_n$ & wave number of Fourier mode $n$ \\
$\lambda_n$ & wavelength of Fourier mode $n$ \\
$D$ & diffusion constant, $D=ka^2$ \\
$r_n$ & eigenvalue of one discrete random-walk time step \\
$J$ & diffusive flux \\
\hline
\end{tabular}
\end{center}
